    last_name = models.CharField(max_length=100)
    surname = models.CharField(max_length=100)
    date_of_birth = models.DateField()
    address = models.CharField(max_length=200)
    
class MedicalCard(models.Model):
    patient = models.ForeignKey(Patient, on_delete=models.CASCADE)
    doctor = models.ForeignKey(User, on_delete=models.CASCADE)
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)
    
class VitalSigns(models.Model):
    card = models.OneToOneField(MedicalCard, on_delete=models.CASCADE)
    temperature = models.DecimalField(max_digits=4, decimal_places=1)
    blood_pressure_systolic = models.IntegerField()
    blood_pressure_diastolic = models.IntegerField()

from rest_framework import serializers
from .models import MedicalCard

class MedicalCardSerializer(serializers.ModelSerializer):
    class Meta:
        model = MedicalCard
        fields = '__all__'
    
    def validate(self, data):
        # Валидация медицинских данных
        if data.get('blood_pressure_systolic', 0) < 0:
